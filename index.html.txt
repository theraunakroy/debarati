  <script>
    const questions = [
      "Do you know you are the cutest girl alive? 🥰",
      "Do you know you make my heart skip a beat? 💓",
      "Do you know I smile like an idiot because of you? 😄",
      "Do you know I want you forever? 💍",
      "Will you be my Valentine, Debarati? 💖"
    ];

    let index = 0;
    const questionEl = document.getElementById("question");
    const noBtn = document.getElementById("noBtn");
    const yesBtn = document.getElementById("yesBtn");
    const questionBox = document.getElementById("questionBox");
    const letter = document.getElementById("letter");

    yesBtn.onclick = () => {
      index++;
      if (index < questions.length) {
        questionEl.innerText = questions[index];
      } else {
        questionBox.style.display = "none";
        letter.style.display = "block";
      }
    };

    noBtn.onmouseover = () => {
      const x = Math.random() * (window.innerWidth - 100);
      const y = Math.random() * (window.innerHeight - 50);
      noBtn.style.left = x + "px";
      noBtn.style.top = y + "px";
    };

    // Floating hearts 💕
    setInterval(() => {
      const heart = document.createElement("div");
      heart.className = "heart";
      heart.innerHTML = "💖";
      heart.style.left = Math.random() * window.innerWidth + "px";
      heart.style.bottom = "0px";
      document.body.appendChild(heart);
      setTimeout(() => heart.remove(), 6000);
    }, 300);
